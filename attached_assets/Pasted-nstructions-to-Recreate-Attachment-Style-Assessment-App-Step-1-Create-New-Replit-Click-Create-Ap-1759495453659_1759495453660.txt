File: shared/schema.ts

import { sql } from "drizzle-orm";
import { pgTable, varchar, jsonb, timestamp, boolean } from "drizzle-orm/pg-core";
import { createInsertSchema } from "drizzle-zod";
import { z } from "zod";
export const assessments = pgTable("assessments", {
  id: varchar("id").primaryKey().default(sql`gen_random_uuid()`),
  responses: jsonb("responses").notNull(),
  result: jsonb("result"),
  shareToken: varchar("share_token"),
  isShared: boolean("is_shared").default(false),
  createdAt: timestamp("created_at").defaultNow().notNull(),
});
export const insertAssessmentSchema = createInsertSchema(assessments).pick({
  responses: true,
});
export const attachmentStyleResultSchema = z.object({
  primaryStyle: z.enum(["secure", "anxious", "avoidant", "fearful"]),
  stylePercentages: z.object({
    secure: z.number(),
    anxious: z.number(),
    avoidant: z.number(),
    fearful: z.number(),
  }),
  description: z.string(),
  strengths: z.array(z.string()),
  growthAreas: z.array(z.string()),
  analysis: z.string(),
});
export type Assessment = typeof assessments.$inferSelect;
export type AttachmentStyleResult = z.infer<typeof attachmentStyleResultSchema>;

Step 10: Create Backend Files
File: server/storage.ts

import { type Assessment, assessments } from "@shared/schema";
import { randomUUID } from "crypto";
import { neon } from "@neondatabase/serverless";
import { drizzle } from "drizzle-orm/neon-http";
import { eq } from "drizzle-orm";
const sql = neon(process.env.DATABASE_URL!);
const db = drizzle(sql);
export const storage = {
  async createAssessment(data: any): Promise<Assessment> {
    const [result] = await db.insert(assessments).values(data).returning();
    return result;
  },
  
  async getAssessment(id: string): Promise<Assessment | undefined> {
    const [result] = await db.select().from(assessments).where(eq(assessments.id, id)).limit(1);
    return result;
  },
  
  async updateAssessmentResult(id: string, result: any): Promise<Assessment | undefined> {
    const [updated] = await db.update(assessments).set({ result }).where(eq(assessments.id, id)).returning();
    return updated;
  },
  
  async enableSharing(id: string): Promise<Assessment | undefined> {
    const shareToken = randomUUID().replace(/-/g, '').substring(0, 16);
    const [updated] = await db.update(assessments).set({ shareToken, isShared: true }).where(eq(assessments.id, id)).returning();
    return updated;
  },
  
  async getSharedAssessment(shareToken: string): Promise<Assessment | undefined> {
    const [result] = await db.select().from(assessments).where(eq(assessments.shareToken, shareToken)).limit(1);
    return result;
  }
};
File: server/services/openai.ts

import OpenAI from "openai";
import type { AttachmentStyleResult } from "@shared/schema";
const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
export async function analyzeAttachmentStyle(responses: Record<string, string>): Promise<AttachmentStyleResult> {
  const prompt = `You are an expert in attachment theory psychology. Analyze these responses and return JSON:
${Object.entries(responses).map(([q, a]) => `${q}: ${a}`).join('\n')}
Return this exact JSON format:
{
  "primaryStyle": "secure" | "anxious" | "avoidant" | "fearful",
  "stylePercentages": { "secure": number, "anxious": number, "avoidant": number, "fearful": number },
  "description": "detailed description",
  "strengths": ["array of strengths"],
  "growthAreas": ["array of growth areas"],
  "analysis": "comprehensive analysis"
}`;
  const response = await openai.chat.completions.create({
    model: "gpt-5",
    messages: [
      { role: "system", content: "You are a compassionate attachment theory expert." },
      { role: "user", content: prompt }
    ],
    response_format: { type: "json_object" },
  });
  return JSON.parse(response.choices[0].message.content || "{}");
}
File: server/routes.ts

import type { Express } from "express";
import { createServer } from "http";
import { storage } from "./storage";
import { insertAssessmentSchema } from "@shared/schema";
import { analyzeAttachmentStyle } from "./services/openai";
export function registerRoutes(app: Express) {
  app.post("/api/assessments", async (req, res) => {
    const data = insertAssessmentSchema.parse(req.body);
    const assessment = await storage.createAssessment(data);
    res.json({ assessmentId: assessment.id });
  });
  app.post("/api/assessments/:id/analyze", async (req, res) => {
    const assessment = await storage.getAssessment(req.params.id);
    if (!assessment) return res.status(404).json({ message: "Not found" });
    
    if (assessment.result) return res.json({ result: assessment.result });
    
    const result = await analyzeAttachmentStyle(assessment.responses as any);
    await storage.updateAssessmentResult(req.params.id, result);
    res.json({ result });
  });
  app.post("/api/assessments/:id/share", async (req, res) => {
    const updated = await storage.enableSharing(req.params.id);
    res.json({ 
      shareToken: updated?.shareToken,
      shareUrl: `${req.protocol}://${req.get('host')}/shared/${updated?.shareToken}`
    });
  });
  app.get("/api/shared/:shareToken", async (req, res) => {
    const assessment = await storage.getSharedAssessment(req.params.shareToken);
    if (!assessment?.isShared) return res.status(404).json({ message: "Not found" });
    res.json({ result: assessment.result });
  });
  return createServer(app);
}
File: server/index.ts

import express from "express";
import { registerRoutes } from "./routes";
import { setupVite, serveStatic } from "./vite";
const app = express();
app.use(express.json());
(async () => {
  const server = registerRoutes(app);
  
  if (process.env.NODE_ENV === "development") {
    const { setupVite } = await import("./vite");
    await setupVite(app, server);
  } else {
    const { serveStatic } = await import("./vite");
    serveStatic(app);
  }
  server.listen(5000, "0.0.0.0", () => {
    console.log("Server on port 5000");
  });
})();
